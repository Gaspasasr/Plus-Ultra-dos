##Accepting Mestizos
country_event = {
	id = 99106
	title = "SPA06.T"
	desc = "SPA06.D"
	
	trigger = {
		SPA = {	
			check_variable = {
			which = Mestizo_ACC
			value = 8
			}
		}
		has_country_effect = mestizo_dilemma
		NOT = { 
			has_global_flag = mestizos_accepted
		}
	}
	
	mean_time_to_happen = { months = 12 }
	
	option = {
		name = "SPA06.A"
		prestige = 30
			add_accepted_culture = mexican
			add_accepted_culture = central_american
			add_accepted_culture = north_andean
			add_accepted_culture = south_andean
			add_accepted_culture = caribeño
			any_country = {
				limit = {
					substate_of = THIS
				}
				all_core = {
					add_core = THIS
					remove_core = MEX
					remove_core = PEU
					remove_core = CLM
					remove_core = GCO
				}
				annex_to = THIS
			}
		remove_country_modifier = mestizo_dilemma
		set_global_flag = mestizos_accepted
	}
}

			SPA = {

				}
			}

					value = 5
				}

			}
		}
